# Test dependencies for the whole repository — scanner, collector AND server.
#
# The suite in ot_scanner/tests/ covers all three, so running it needs the
# server's dependencies too. Without them the server tests do not fail, they
# ERROR at import, which reads as a broken checkout rather than a missing
# install.
pytest>=7.0

# Server (ot_server/). Kept here rather than referenced with -r so a single
# `pip install -r requirements-dev.txt` from ot_scanner/ sets up everything.
fastapi>=0.110
psycopg[binary]>=3.1

# starlette's TestClient imports httpx, and neither fastapi nor starlette
# depends on it. Undeclared, the API tests do not skip — they RAISE at import,
# and three commits landed on main red for exactly that reason before anyone
# looked. tests/test_dependencies.py now checks both halves: that a client is
# named here, and that TestClient really imports wherever fastapi does.
#
# Coming migration: starlette warns that using httpx here is deprecated in
# favour of httpx2. Left on httpx until httpx2 is the version fastapi's own
# matrix tests against, so the pin does not run ahead of the framework.
httpx>=0.27

# The fleet CA (Phase 6, decision Q4). Server-side only: the collector shells
# out to openssl so manifest.RUNTIME_REQUIRES stays at dpkt alone, and
# tests/test_collector_manifest.py fails if that changes.
# The floor is 42.0 for a reason: `not_valid_before_utc` /
# `not_valid_after_utc` arrive there, and the naive-datetime accessors they
# replace return values that compare wrongly against the timezone-aware
# timestamps PostgreSQL hands back. tests/test_dependencies.py fails with that
# sentence rather than letting a bad resolution surface as an AttributeError
# inside ca.sign().
#
# No ceiling is needed for the 3.8 leg of the CI matrix, and this was checked
# rather than assumed: 48.0 raised requires-python to >=3.9, so pip on 3.8
# resolves to 47.0.0 and stays there. That release ships a cp38-abi3 wheel (no
# Rust toolchain on the runner) and still carries both accessors.
cryptography>=42.0

# Brand assets. tools/build_brand_assets.py derives console/public/otsec-*.png
# from docs/brand/otsec-master.png, and tests/test_branding.py rebuilds them to
# check the committed files still match the master.
#
# This is declared here and NOT in requirements.txt on purpose: the container
# never builds assets, the derived PNGs are committed, and a deployment needs
# nothing but the repository. It is declared at all — rather than left to
# importorskip — because a test that silently skips reads exactly like a test
# that passed, which is the same reasoning as OT_CONSOLE_REQUIRED above.
#
# No ceiling, and the 3.8 leg was checked rather than assumed: Pillow 11 raised
# requires-python to >=3.9, so pip on 3.8 resolves to 10.4.0, which ships a
# cp38 wheel and carries every call this tool makes (quantize with MEDIANCUT
# and dither=NONE, LANCZOS resampling). The drift check compares PIXELS rather
# than encoded bytes precisely so this version spread cannot turn a runner
# upgrade into a red build.
Pillow>=10.0
