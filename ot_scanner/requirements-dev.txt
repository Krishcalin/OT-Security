# Test dependencies for the whole repository — scanner, collector AND server.
#
# The suite in ot_scanner/tests/ covers all three, so running it needs the
# server's dependencies too. Without them the server tests do not fail, they
# ERROR at import, which reads as a broken checkout rather than a missing
# install.
pytest>=7.0

# Server (ot_server/). Kept here rather than referenced with -r so a single
# `pip install -r requirements-dev.txt` from ot_scanner/ sets up everything.
fastapi>=0.110
psycopg[binary]>=3.1

# starlette's TestClient imports httpx, and neither fastapi nor starlette
# depends on it. Without this the API tests do not skip — they ERROR at import,
# which on a summary page looks like an infrastructure problem rather than an
# untested estate plane. It is installed here explicitly rather than relied on
# as somebody else's transitive dependency.
httpx>=0.27

# The fleet CA (Phase 6, decision Q4). Server-side only: the collector shells
# out to openssl so manifest.RUNTIME_REQUIRES stays at dpkt alone, and
# tests/test_collector_manifest.py fails if that changes.
# The floor is 42.0 for a reason: `not_valid_before_utc` /
# `not_valid_after_utc` arrive there, and the naive-datetime accessors
# they replace return values that compare wrongly against the
# timezone-aware timestamps PostgreSQL hands back.
cryptography>=42.0
